cmake_minimum_required(VERSION 3.30)
project(LWSSettings VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LWSSETTINGS_BUILD_TESTS "Build settings tests" ${PROJECT_IS_TOP_LEVEL})
option(LWSSETTINGS_BUILD_DEMO "Build settings demo" ${PROJECT_IS_TOP_LEVEL})
option(LWSSETTINGS_ENABLE_EVENT_DEBUGGER "Enable the demo event window and F8 shortcut" ON)
set(LWSSETTINGS_LWS_SOURCE_DIR "" CACHE PATH "LWS source directory")
set(LWSSETTINGS_JSON_SOURCE_DIR "" CACHE PATH "nlohmann/json source directory")
set(LWSSETTINGS_JSON_INCLUDE_DIR "" CACHE PATH "nlohmann/json include directory (header-only, bypasses find_package)")
if(NOT TARGET LWSLib)
    if(NOT LWSSETTINGS_LWS_SOURCE_DIR)
        message(FATAL_ERROR "Provide LWSLib or set LWSSETTINGS_LWS_SOURCE_DIR")
    endif()
    set(LWS_BUILD_TESTS OFF CACHE BOOL "" )
    add_subdirectory("${LWSSETTINGS_LWS_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/dependencies/LWS")
endif()
get_target_property(_lws_definitions LWSLib INTERFACE_COMPILE_DEFINITIONS)
if(NOT WIN32 AND NOT "LWS_PLATFORM_WAYLAND" IN_LIST _lws_definitions)
    message(FATAL_ERROR "LWSSettings requires the LWS Windows or Wayland backend")
endif()
if(NOT TARGET nlohmann_json::nlohmann_json)
    if(LWSSETTINGS_JSON_INCLUDE_DIR)
        add_library(nlohmann_json::nlohmann_json INTERFACE IMPORTED)
        set_target_properties(nlohmann_json::nlohmann_json PROPERTIES
            INTERFACE_INCLUDE_DIRECTORIES "${LWSSETTINGS_JSON_INCLUDE_DIR}")
    elseif(LWSSETTINGS_JSON_SOURCE_DIR)
        set(JSON_BuildTests OFF CACHE BOOL "")
        add_subdirectory("${LWSSETTINGS_JSON_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/dependencies/json")
    else()
        find_package(nlohmann_json 3.12 REQUIRED)
    endif()
endif()
# Keep one consumable settings target until a headless consumer needs a core target.
# Splitting document/storage from UI would otherwise add packaging/dependency policy
# without a current consumer benefit; the source boundaries already permit that split.
add_library(LWSSettings STATIC src/SettingsDocument.cpp src/SettingsStorage.cpp src/SettingsView.cpp src/SettingsHeader.cpp src/SettingsEditor.cpp src/ThemeStorage.cpp src/ThemeFields.cpp src/ThemeEditor.cpp src/ThemeSession.cpp src/ThemeSelector.cpp src/SettingsIcon.cpp)
add_library(LWSSettings::LWSSettings ALIAS LWSSettings)
target_compile_features(LWSSettings PUBLIC cxx_std_26)
target_include_directories(LWSSettings PUBLIC include PRIVATE src)
target_link_libraries(LWSSettings PUBLIC LWSUI::LWSUI nlohmann_json::nlohmann_json)
if(NOT WIN32)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SettingsPng REQUIRED IMPORTED_TARGET cairo)
    target_link_libraries(LWSSettings PRIVATE PkgConfig::SettingsPng)
endif()
if(LWSSETTINGS_BUILD_DEMO)
    add_executable(LWSSettingsDemo examples/main.cpp)
    target_link_libraries(LWSSettingsDemo PRIVATE LWSSettings)
    target_compile_definitions(LWSSettingsDemo PRIVATE
        LWSSETTINGS_ENABLE_EVENT_DEBUGGER=$<BOOL:${LWSSETTINGS_ENABLE_EVENT_DEBUGGER}>)
    if(WIN32)
        set_target_properties(LWSSettingsDemo PROPERTIES WIN32_EXECUTABLE TRUE)
        if(MSVC)
            target_link_options(LWSSettingsDemo PRIVATE /ENTRY:mainCRTStartup)
        elseif(MINGW)
            target_link_options(LWSSettingsDemo PRIVATE -Wl,--entry,mainCRTStartup)
        endif()
    endif()
    # Launching from the executable directory must work with custom runtime output
    # paths and multi-config generators, not just from the top-level build folder.
    # Settings defaults live in the template. Saved values and theme overrides are user-owned.
    file(GENERATE OUTPUT "$<TARGET_FILE_DIR:LWSSettingsDemo>/settings.json"
        INPUT "${CMAKE_CURRENT_SOURCE_DIR}/examples/settings.json"
        TARGET LWSSettingsDemo NEWLINE_STYLE CRLF)
    file(GENERATE OUTPUT "$<TARGET_FILE_DIR:LWSSettingsDemo>/oiv/settings.json"
        INPUT "${CMAKE_CURRENT_SOURCE_DIR}/examples/oiv/settings.json"
        TARGET LWSSettingsDemo NEWLINE_STYLE CRLF)
endif()
if(LWSSETTINGS_BUILD_TESTS)
    include(CTest)
    if(NOT TARGET Catch2::Catch2WithMain)
        include(FetchContent)
        FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.7.1 GIT_SHALLOW TRUE)
        FetchContent_MakeAvailable(Catch2)
    endif()
    add_executable(LWSSettingsTests tests/settings.cpp)
    target_include_directories(LWSSettingsTests PRIVATE src examples)
    target_compile_definitions(LWSSettingsTests PRIVATE LWSSETTINGS_EXAMPLES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples")
    target_link_libraries(LWSSettingsTests PRIVATE LWSSettings Catch2::Catch2WithMain)
    target_compile_definitions(LWSSettingsTests PRIVATE
        LWSSETTINGS_ENABLE_EVENT_DEBUGGER=$<BOOL:${LWSSETTINGS_ENABLE_EVENT_DEBUGGER}>)
    get_target_property(_catch_source Catch2 SOURCE_DIR)
    list(APPEND CMAKE_MODULE_PATH "${_catch_source}/../extras")
    include(Catch)
    catch_discover_tests(LWSSettingsTests)
    if(LWSSETTINGS_BUILD_DEMO)
        find_package(Python3 COMPONENTS Interpreter QUIET)
        if(Python3_Interpreter_FOUND)
            add_test(NAME LWSSettingsDemoStartup COMMAND "${Python3_EXECUTABLE}"
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/demo_startup.py"
                "$<TARGET_FILE:LWSSettingsDemo>" "${CMAKE_CURRENT_SOURCE_DIR}/examples"
                "${CMAKE_CURRENT_BINARY_DIR}/demo-startup-tests" "$<BOOL:${LWSSETTINGS_ENABLE_EVENT_DEBUGGER}>")
            set_tests_properties(LWSSettingsDemoStartup PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE)
        endif()
    endif()
endif()
