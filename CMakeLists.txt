cmake_minimum_required(VERSION 3.30)
project(LWSSettings VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LWSSETTINGS_BUILD_TESTS "Build settings tests" ${PROJECT_IS_TOP_LEVEL})
option(LWSSETTINGS_BUILD_DEMO "Build settings demo" ${PROJECT_IS_TOP_LEVEL})
set(LWSSETTINGS_LWS_SOURCE_DIR "" CACHE PATH "LWS source directory")
set(LWSSETTINGS_JSON_SOURCE_DIR "" CACHE PATH "nlohmann/json source directory")
if(NOT TARGET LWSLib)
    if(NOT LWSSETTINGS_LWS_SOURCE_DIR)
        message(FATAL_ERROR "Provide LWSLib or set LWSSETTINGS_LWS_SOURCE_DIR")
    endif()
    set(LWS_BUILD_EXAMPLES OFF CACHE BOOL "" )
    set(LWS_BUILD_TESTS OFF CACHE BOOL "" )
    add_subdirectory("${LWSSETTINGS_LWS_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/dependencies/LWS")
endif()
get_target_property(_lws_definitions LWSLib INTERFACE_COMPILE_DEFINITIONS)
if(NOT WIN32 AND NOT "LWS_PLATFORM_WAYLAND" IN_LIST _lws_definitions)
    message(FATAL_ERROR "LWSSettings requires the LWS Windows or Wayland backend")
endif()
if(NOT TARGET nlohmann_json::nlohmann_json)
    if(LWSSETTINGS_JSON_SOURCE_DIR)
        set(JSON_BuildTests OFF CACHE BOOL "")
        add_subdirectory("${LWSSETTINGS_JSON_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/dependencies/json")
    else()
        find_package(nlohmann_json 3.12 REQUIRED)
    endif()
endif()
add_library(LWSSettings STATIC src/SettingsPage.cpp src/Viewer.cpp src/Canvas.cpp src/Controls.cpp)
add_library(LWSSettings::LWSSettings ALIAS LWSSettings)
target_compile_features(LWSSettings PUBLIC cxx_std_26)
target_include_directories(LWSSettings PUBLIC include PRIVATE src)
target_link_libraries(LWSSettings PUBLIC LWSLib nlohmann_json::nlohmann_json LLUtils::LLUtils)
if(WIN32)
    target_compile_definitions(LWSSettings PUBLIC NOMINMAX UNICODE _UNICODE)
    target_link_libraries(LWSSettings PRIVATE d2d1 dwrite windowscodecs)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SettingsText REQUIRED IMPORTED_TARGET pangocairo fontconfig)
    target_link_libraries(LWSSettings PRIVATE PkgConfig::SettingsText)
endif()
if(LWSSETTINGS_BUILD_DEMO)
    add_executable(LWSSettingsDemo examples/main.cpp)
    target_link_libraries(LWSSettingsDemo PRIVATE LWSSettings)
    configure_file(examples/settings.json settings.json COPYONLY)
    configure_file(examples/showcase-values.json showcase-values.json COPYONLY)
    file(GENERATE OUTPUT "$<TARGET_FILE_DIR:LWSSettingsDemo>/oiv/settings.json"
        INPUT "${CMAKE_CURRENT_SOURCE_DIR}/examples/oiv/settings.json"
        TARGET LWSSettingsDemo NEWLINE_STYLE CRLF)
    file(GENERATE OUTPUT "$<TARGET_FILE_DIR:LWSSettingsDemo>/oiv/showcase-values.json"
        INPUT "${CMAKE_CURRENT_SOURCE_DIR}/examples/oiv/values.json"
        TARGET LWSSettingsDemo NEWLINE_STYLE CRLF)
endif()
if(LWSSETTINGS_BUILD_TESTS)
    include(CTest)
    if(NOT TARGET Catch2::Catch2WithMain)
        include(FetchContent)
        FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.7.1 GIT_SHALLOW TRUE)
        FetchContent_MakeAvailable(Catch2)
    endif()
    add_executable(LWSSettingsTests tests/settings.cpp)
    target_include_directories(LWSSettingsTests PRIVATE src examples)
    target_compile_definitions(LWSSettingsTests PRIVATE LWSSETTINGS_EXAMPLES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples")
    target_link_libraries(LWSSettingsTests PRIVATE LWSSettings Catch2::Catch2WithMain)
    get_target_property(_catch_source Catch2 SOURCE_DIR)
    list(APPEND CMAKE_MODULE_PATH "${_catch_source}/../extras")
    include(Catch)
    catch_discover_tests(LWSSettingsTests)
endif()
